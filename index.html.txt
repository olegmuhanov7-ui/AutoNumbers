<script>
  // ===== Telegram WebApp =====
  const tg = window.Telegram?.WebApp;
  if (tg) {
    tg.ready();
    tg.expand();
  }

  // ===== Константы =====
  const STORAGE_KEY = 'autonumbers_state_v2';
  const ALLOWED_LETTERS = ['А', 'В', 'Е', 'К', 'М', 'Н', 'О', 'Р', 'С', 'Т', 'У', 'Х'];
  const REGIONS = [
    '77', '99', '97', '177', '199', '197',
    '50', '90', '150', '190', '98', '178',
    '78', '47', '23', '161', '61', '34',
    '36', '52', '66', '74', '102', '116',
    '123', '124', '125', '126', '134', '136',
    '138', '142', '152', '154', '156', '159',
    '164', '166', '173', '174', '175', '176',
    '179', '182', '186', '187', '188', '189',
    '191', '192', '193', '194', '195', '196',
    '198'
  ];
  const PREMIUM_REGIONS = ['77', '99', '97', '177', '199', '197'];

  // ===== Состояние =====
  let state = {
    balance: 0,
    savedNumbers: [],
    currentNumber: null,
    currentPrice: 0,
    currentRarity: null
  };

  function loadState() {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (raw) state = { ...state, ...JSON.parse(raw) };
    } catch (e) { console.warn('load error', e); }
  }

  function saveState() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify({
        balance: state.balance,
        savedNumbers: state.savedNumbers
      }));
    } catch (e) { console.warn('save error', e); }
  }

  // ===== Генерация =====
  function generateNumber() {
    const rand = (max) => Math.floor(Math.random() * max);
    const randLetter = () => ALLOWED_LETTERS[rand(ALLOWED_LETTERS.length)];
    const randDigit = () => rand(10);

    const letter1 = randLetter();
    const digits = `${randDigit()}${randDigit()}${randDigit()}`;
    const letter2 = randLetter();
    const letter3 = randLetter();
    const region = REGIONS[rand(REGIONS.length)];

    return `${letter1}${digits}${letter2}${letter3} ${region}`;
  }

  function isUnique(number) {
    return !state.savedNumbers.includes(number);
  }

  function getUniqueNumber() {
    for (let i = 0; i < 200; i++) {
      const num = generateNumber();
      if (isUnique(num)) return num;
    }
    return null;
  }

  // ===== Оценка =====
  function calculatePrice(number) {
    const match = number.match(/^([А-Я])(\d{3})([А-Я]{2})\s(\d{2,3})$/);
    if (!match) return { price: 10, rarity: 'common' };

    const [, letter1, digits, letters23, region] = match;
    const allLetters = letter1 + letters23;
    const digitArr = digits.split('').map(Number);

    let price = 50;
    let score = 0;

    // Повтор букв
    if (letters23[0] === letters23[1]) { price += 200; score += 2; }
    if (letter1 === letters23[0]) { price += 150; score += 1; }

    // Повтор цифр
    if (digits[0] === digits[1] && digits[1] === digits[2]) {
      price += 500; score += 4;
    } else if (digits[0] === digits[1] || digits[1] === digits[2] || digits[0] === digits[2]) {
      price += 100; score += 1;
    }

    // Последовательности
    if (digitArr[1] - digitArr[0] === 1 && digitArr[2] - digitArr[1] === 1) {
      price += 300; score += 2;
    }
    if (digitArr[0] - digitArr[1] === 1 && digitArr[1] - digitArr[2] === 1) {
      price += 300; score += 2;
    }

    // Круглые
    if (['001', '100', '700'].includes(digits)) { price += 250; score += 2; }

    // Зеркальные буквы
    if (letter1 === letters23[1]) { price += 150; score += 1; }

    // Премиум регион
    if (PREMIUM_REGIONS.includes(region)) { price += 100; score += 1; }

    // Все буквы одинаковые (ААА) — легендарно
    if (allLetters[0] === allLetters[1] && allLetters[1] === allLetters[2]) {
      price += 1000; score += 5;
    }

    // Легендарные серии
    const LEGENDARY = ['АМР', 'ЕКХ', 'АМО', 'СОО', 'ООО', 'КОО', 'ВОО', 'МММ', 'ХКХ'];
    if (LEGENDARY.includes(allLetters)) { price += 2000; score += 6; }

    // Определяем редкость
    let rarity = 'common';
    if (score >= 7) rarity = 'legendary';
    else if (score >= 4) rarity = 'epic';
    else if (score >= 2) rarity = 'rare';
    else if (score >= 1) rarity = 'uncommon';

    return { price, rarity };
  }

  // ===== UI =====
  const display = document.getElementById('display');
  const reel = document.getElementById('reel');
  const glow = document.getElementById('glow');
  const rarityBadge = document.getElementById('rarityBadge');
  const spinBtn = document.getElementById('spinBtn');
  const actions = document.getElementById('actions');
  const saveBtn = document.getElementById('saveBtn');
  const sellBtn = document.getElementById('sellBtn');
  const balanceEl = document.getElementById('balance');
  const savedCountEl = document.getElementById('savedCount');
  const collectionEl = document.getElementById('collection');

  const RARITY_STYLES = {
    common:    { label: '',            color: '#888',    glow: 'rgba(79,195,247,0.15)' },
    uncommon:  { label: 'Необычный',   color: '#4fc3f7', glow: 'rgba(79,195,247,0.25)' },
    rare:      { label: 'Редкий',      color: '#b388ff', glow: 'rgba(179,136,255,0.3)' },
    epic:      { label: 'Эпический',   color: '#ff4081', glow: 'rgba(255,64,129,0.35)' },
    legendary: { label: 'Легендарный', color: '#ffd600', glow: 'rgba(255,214,0,0.4)' }
  };

  function renderPlate(number) {
    const match = number.match(/^([А-Я])(\d{3})([А-Я]{2})\s(\d{2,3})$/);
    if (!match) return `<span class="main">${number}</span>`;
    const [, l1, d, l23, region] = match;
    return `
      <div class="plate">
        <span class="main">${l1}${d}${l23}</span>
        <span class="region">
          <span class="flag">RUS</span>
          <span class="code">${region}</span>
        </span>
      </div>
    `;
  }

  function setReelColor(rarity) {
    const style = RARITY_STYLES[rarity] || RARITY_STYLES.common;
    reel.style.borderColor = style.color + '66';
    reel.style.boxShadow = `inset 0 0 30px rgba(0,0,0,0.6), 0 0 40px ${style.glow}`;
    glow.style.background = `radial-gradient(circle, ${style.glow} 0%, transparent 70%)`;

    if (style.label) {
      rarityBadge.textContent = style.label;
      rarityBadge.style.color = style.color;
      rarityBadge.style.background = style.color + '22';
      rarityBadge.classList.add('show');
    } else {
      rarityBadge.classList.remove('show');
    }
  }

  function resetReelColor() {
    reel.style.borderColor = '#2a2a40';
    reel.style.boxShadow = 'inset 0 0 30px rgba(0,0,0,0.6), 0 0 40px rgba(79,195,247,0.1)';
    glow.style.background = 'radial-gradient(circle, rgba(79, 195, 247, 0.15) 0%, transparent 70%)';
    rarityBadge.classList.remove('show');
  }

  function updateUI() {
    balanceEl.textContent = state.balance;
    savedCountEl.textContent = state.savedNumbers.length;

    if (state.savedNumbers.length === 0) {
      collectionEl.innerHTML = '<span class="empty">Пока пусто</span>';
    } else {
      collectionEl.innerHTML = state.savedNumbers
        .map(n => {
          const { rarity } = calculatePrice(n);
          return `<span class="chip rarity-${rarity}">${n}</span>`;
        })
        .join('');
    }
  }

  // ===== Крутка =====
  let spinning = false;

  function spin() {
    if (spinning) return;

    const finalNumber = getUniqueNumber();
    if (!finalNumber) {
      tg?.showAlert?.('Все номера закончились!') || alert('Все номера закончились!');
      return;
    }

    spinning = true;
    spinBtn.disabled = true;
    actions.classList.remove('show');
    state.currentNumber = null;
    state.currentPrice = 0;
    resetReelColor();

    display.classList.add('spinning');
    display.classList.remove('win');

    const spinDuration = 1800;
    const startTime = Date.now();
    const interval = 60;

    const spinTimer = setInterval(() => {
      display.innerHTML = renderPlate(generateNumber());

      if (Date.now() - startTime > spinDuration) {
        clearInterval(spinTimer);
        finishSpin(finalNumber);
      }
    }, interval);
  }

  function finishSpin(number) {
    const { price, rarity } = calculatePrice(number);

    display.classList.remove('spinning');
    display.innerHTML = renderPlate(number);
    display.classList.add('win');

    setReelColor(rarity);

    state.currentNumber = number;
    state.currentPrice = price;
    state.currentRarity = rarity;

    spinning = false;
    spinBtn.disabled = false;
    actions.classList.add('show');

    tg?.HapticFeedback?.impactOccurred?.(rarity === 'legendary' ? 'heavy' : 'medium');
  }

  // ===== Действия =====
  function saveNumber() {
    if (!state.currentNumber) return;

    state.savedNumbers.push(state.currentNumber);
    state.currentNumber = null;
    state.currentPrice = 0;
    state.currentRarity = null;

    saveState();
    updateUI();

    actions.classList.remove('show');
    display.classList.remove('win');
    display.innerHTML = renderPlate('А123ВС 77');
    resetReelColor();

    tg?.HapticFeedback?.notificationOccurred?.('success');
  }

  function sellNumber() {
    if (!state.currentNumber) return;

    const price = state.currentPrice;
    state.balance += price;
    state.currentNumber = null;
    state.currentPrice = 0;
    state.currentRarity = null;

    saveState();
    updateUI();

    actions.classList.remove('show');
    display.classList.remove('win');
    display.innerHTML = renderPlate('А123ВС 77');
    resetReelColor();

    tg?.HapticFeedback?.notificationOccurred?.('success');

    const msg = `Продано за ${price} ₽`;
    if (tg?.showAlert) tg.showAlert(msg);
  }

  // ===== Частицы =====
  function createParticles() {
    const container = document.getElementById('particles');
    const colors = ['#4fc3f7', '#b388ff', '#00e676'];

    for (let i = 0; i < 30; i++) {
      const span = document.createElement('span');
      span.style.left = Math.random() * 100 + '%';
      span.style.animationDuration = (8 + Math.random() * 12) + 's';
      span.style.animationDelay = (Math.random() * 10) + 's';
      span.style.opacity = 0.2 + Math.random() * 0.4;
      span.style.background = colors[Math.floor(Math.random() * colors.length)];
      container.appendChild(span);
    }
  }

  // ===== Init =====
  function init() {
    loadState();
    updateUI();
    createParticles();
    display.innerHTML = renderPlate('А123ВС 77');

    spinBtn.addEventListener('click', spin);
    saveBtn.addEventListener('click', saveNumber);
    sellBtn.addEventListener('click', sellNumber);
  }

  init();
</script>
